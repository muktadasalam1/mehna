# runtime.py - Mehna Job Platform Runner with Cloudflare Tunnel (Quiet Mode)
import os
import re
import sys
import time
import threading
import subprocess
from app import app, socketio, seed_data

# Disable Flask request logs
import logging
log = logging.getLogger('werkzeug')
log.setLevel(logging.ERROR)

def find_cloudflared():
    """Find cloudflared executable in project directory"""
    for name in ['cloudflared.exe', 'cloudflared-windows-amd64.exe']:
        if os.path.exists(name):
            return name
    return None

def start_cloudflare_tunnel():
    """Start Cloudflare Tunnel automatically"""
    time.sleep(2)
    
    tunnel_exe = find_cloudflared()
    
    if not tunnel_exe:
        print("[!] cloudflared not found - running locally only")
        return
    
    try:
        tunnel_process = subprocess.Popen(
            [tunnel_exe, 'tunnel', '--url', 'http://localhost:5000', '--no-autoupdate'],
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            text=True,
            bufsize=1
        )
        
        for line in tunnel_process.stdout:
            if 'trycloudflare.com' in line and 'https://' in line:
                match = re.search(r'https://[a-z0-9\-]+\.trycloudflare\.com', line)
                if match:
                    print("\n" + "="*50)
                    print("PUBLIC URL: " + match.group(0))
                    print("="*50)
                    print("Demo Accounts:")
                    print("  Seeker:  seeker@mehna.com / 123456")
                    print("  Employer: employer@mehna.com / 123456")
                    print("="*50 + "\n")
                    break
                    
    except Exception as e:
        print(f"[!] Cloudflare Tunnel failed: {e}")

if __name__ == '__main__':
    seed_data()
    threading.Thread(target=start_cloudflare_tunnel, daemon=True).start()
    print("Server running on http://127.0.0.1:5000\n")
    socketio.run(app, debug=False, allow_unsafe_werkzeug=True)
